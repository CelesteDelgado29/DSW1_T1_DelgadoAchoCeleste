GO
USE BDBIBLIOTECA2026
GO

CREATE TABLE dbo.Libros
(
    Id INT IDENTITY(1,1) PRIMARY KEY,
    NombreLibro VARCHAR(120) NOT NULL,
    Autor VARCHAR(100) NOT NULL,
    NombreEditorial VARCHAR(100) NOT NULL,
    ISDN VARCHAR(20) NOT NULL
)
GO

CREATE PROCEDURE usp_ListarLibros
AS
BEGIN
    SELECT Id, NombreLibro, Autor, NombreEditorial, ISDN
    FROM dbo.Libros
END
GO

CREATE PROCEDURE usp_RegistrarLibro
    @NombreLibro VARCHAR(120),
    @Autor VARCHAR(100),
    @NombreEditorial VARCHAR(100),
    @ISDN VARCHAR(20)
AS
BEGIN
    INSERT INTO dbo.Libros (NombreLibro, Autor, NombreEditorial, ISDN)
    VALUES (@NombreLibro, @Autor, @NombreEditorial, @ISDN)
END
GO

CREATE PROCEDURE usp_ActualizarLibro
    @Id INT,
    @NombreLibro VARCHAR(120),
    @Autor VARCHAR(100),
    @NombreEditorial VARCHAR(100),
    @ISDN VARCHAR(20)
AS
BEGIN
    UPDATE dbo.Libros
    SET NombreLibro = @NombreLibro,
        Autor = @Autor,
        NombreEditorial = @NombreEditorial,
        ISDN = @ISDN
    WHERE Id = @Id
END
GO

CREATE PROCEDURE usp_EliminarLibro
    @Id INT
AS
BEGIN
    DELETE FROM dbo.Libros WHERE Id = @Id
END
GO

CREATE PROCEDURE usp_ListarLibrosPorAutor
    @Autor VARCHAR(100)
AS
BEGIN
    SELECT Id, NombreLibro, Autor, NombreEditorial, ISDN
    FROM dbo.Libros
    WHERE Autor LIKE '%' + @Autor + '%'
END
GO
